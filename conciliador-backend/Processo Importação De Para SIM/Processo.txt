drop table de_para_sim;
CREATE TABLE de_para_sim (
    empresa      int4 not null,
    centro       varchar(04) not null,
    area         varchar(04) not null,
    local        varchar(04) not null,
    divisao      varchar(04) not null,
    organizacao  varchar(10)  default '',
    razao        varchar(70),
    uf           char(02),
    cgc          varchar(25),
    sic          int4,
    PRIMARY KEY(empresa,divisao) 
);
